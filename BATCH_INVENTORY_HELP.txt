usage: batch_inventory [options]

positional arguments:
  cells_dir             Directory for per-FOV cell CSV files.
  cell_layers_dir       Directory for per-FOV cell-layer CSV files.
  foci_dir              Directory for per-FOV focus CSV files.

options:
  -h, --help            show this help message and exit
  -i INCLUDE [INCLUDE ...], --include INCLUDE [INCLUDE ...]
                        Include source ND2 or TIFF files using one or more
                        glob patterns. The option may be passed multiple
                        times; default: ['**/*.nd2'].
  -e EXCLUDE [EXCLUDE ...], --exclude EXCLUDE [EXCLUDE ...]
                        Exclude files or directories using one or more glob
                        patterns; default: [].
  --root ROOT           Root directory against which include and exclude
                        patterns are evaluated; default: '.'.
  --detection-threshold-method DETECTION_THRESHOLD_METHOD
                        What method to use to determine the intensity
                        threshold for focus detection. 'absolute' will use the
                        value provided for --absolute-detection-threshold, and
                        'IQR_multiple' will use the median plus a multiple of
                        the interquartile range of the pixel intensities
                        inside cell boundaries; default: IQR_multiple
  --absolute-detection-threshold ABSOLUTE_DETECTION_THRESHOLD
                        What absolute intensity value a putative focus must
                        surpass in order to be detected; default: 1.0.
  --IQR-multiple IQR_MULTIPLE
                        Multiple of the IQR above the median fluorescence
                        intensity inside the foreground (all cells) which a
                        focus must surpass to qualify for detection. Decrease
                        this to detect dimmer foci; default: 2.0.
  --detect-min-sigma DETECT_MIN_SIGMA
                        Minimum Gaussian blur kernel standard deviation used
                        for Laplacian of Gaussian foci detection. Decrease
                        this to detect smaller foci; default: 2.0
  --detect-max-sigma DETECT_MAX_SIGMA
                        Maximum Gaussian blur kernel standard deviation used
                        for Laplacian of Gaussian foci detection. Increase
                        this to detect larger foci; default: 3.8.
  --detect-num-sigma DETECT_NUM_SIGMA
                        Number of Gaussian blur kernel standard deviations
                        used for Laplacian of Gaussian foci detection.
                        Increase this to differentiate foci more precisely as
                        to their size; default: 20.
  --overlap OVERLAP     Maximum percent overlap between two foci before the
                        smaller is discarded; default: 0.75.
  --expand-by EXPAND_BY
                        Distance (in pixels) to expand each cell label in
                        every direction by for the purpose of assigning foci
                        to it. Increase this to detect foci further outside
                        the edge of a cell label; default: 2
  --plot-crop-margin PLOT_CROP_MARGIN
                        Margin (in pixels) to show around the cell contour for
                        the stepwise plots; default: 3.
  --bg-sub-radius BG_SUB_RADIUS
                        Width (in pixels) of the kernel for background
                        subtraction. Increase this to take into account
                        further out pixels in assigning the background
                        intensity for a given pixel; default: 1.5
  --denoise-radius DENOISE_RADIUS
                        Width (in pixels) of the kernel for denoising.
                        Increase this to smooth features on a larger scale in
                        the denoising stage; default: 2.0.
  --denoise-amount DENOISE_AMOUNT
                        Magnitude of the denoise operation. Increase this to
                        smooth noise features to a greater extent; default:
                        6.0.
  --gaussian-sigma GAUSSIAN_SIGMA
                        Standard deviation of the Gaussian blur kernel.
                        Increase this to blur together features that are
                        farther apart; default: 1.2.
  --save-stepwise-figures SAVE_STEPWISE_FIGURES
                        Directory for per-cell stepwise figures. The source
                        directory structure relative to --root is preserved.
  --max-workers MAX_WORKERS
                        Number of worker threads; set to 0 to use the main
                        thread. Default: 5.
